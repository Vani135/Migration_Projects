          rule_key      = os.environ['RULE_KEY']
          rule_name     = os.environ['RULE_NAME']
          vpc_key       = os.environ['VPC_KEY']

          def hcl_list(items):
              if not items:
                  return '[]'
              quoted = ', '.join(f'"{i}"' for i in items)
              return f'[{quoted}]'

          def hcl_action_block(action, proto, port_list):
              ports_hcl = hcl_list(port_list)
              return (

              )

          allow_block = hcl_action_block('allow', protocol, ports) if rule_action == 'allow' else ''
          deny_block  = hcl_action_block('deny',  protocol, ports) if rule_action == 'deny'  else ''
          log_block   = '    log_config {\n      metadata = "INCLUDE_ALL_METADATA"\n    }' if enable_log else ''

          desc = description or f"Firewall rule {rule_name} for {app_name} ({team_name})"

          rule_entry_lines = [

              f'    project                   = "{os.environ["PROJECT_ID"]}"',

          rule_entry_lines.append('  }')
          rule_entry = '\n'.join(rule_entry_lines)

          def insert_into_map(content, map_name, new_entry, key):
              if re.search(rf'^\s+{re.escape(key)}\s*=', content, re.MULTILINE):
                  print(f'  {key} already in {map_name}')
                  return content
              single = re.compile(rf'^([ \t]*{re.escape(map_name)}[ \t]*=[ \t]*\{{[ \t]*\}})', re.MULTILINE)
              if single.search(content):
                  content = single.sub(f'{map_name} = {{\n{new_entry}\n}}', content)
                  print(f'  Added {key} to {map_name} (expanded empty map)')
                  return content
              if not re.search(rf'^\s*{re.escape(map_name)}\s*=\s*\{{', content, re.MULTILINE):
                  content = content.rstrip('\n') + f'\n\n{map_name} = {{\n{new_entry}\n}}\n'
                  print(f'  Added {key} to {map_name} (new map)')
                  return content
              lines = content.splitlines(keepends=True)
              in_map, depth = False, 0
              for i, line in enumerate(lines):
                  stripped = line.strip()
                  if not in_map:
                      if re.match(rf'\s*{re.escape(map_name)}\s*=\s*\{{', line):
                          in_map = True
                          depth = stripped.count('{') - stripped.count('}')
                  else:
                      depth += stripped.count('{') - stripped.count('}')
                      if depth == 0:
                          lines.insert(i, new_entry + '\n')
                          print(f'  Added {key} to {map_name}')
                          break
              return ''.join(lines)

          content = insert_into_map(content, 'firewall_rules', rule_entry, rule_key)

            MAX_RETRIES=3
            RETRY_COUNT=0
            while [ $RETRY_COUNT -lt $MAX_RETRIES ]; do
              # FIX 7: Only attempt rebase abort if pull --rebase itself failed

                if git push; then
                  echo "Pushed terraform.tfvars update."
